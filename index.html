Google Sheets Integration Setup

Automatically creates 3 live breakdown tabs with totals & calculations.

Apps Script Code (Creates 3 Automated Sheet Tabs) 
function doGet() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName('Expenses') || ss.insertSheet('Expenses');
  ensureSheets(ss);

  var rows = sheet.getDataRange().getValues();
  var data = [];
  for (var i = 1; i < rows.length; i++) {
    if (rows[i][0]) {
      data.push({
        id: String(rows[i][0]),
        date: String(rows[i][1]),
        title: String(rows[i][2]),
        amount: Number(rows[i][3]),
        paidBy: String(rows[i][4]),
        involved: String(rows[i][5]).split(',').map(function(s){ return s.trim(); }).filter(Boolean),
        createdBy: String(rows[i][6] || rows[i][4])
      });
    }
  }
  return ContentService.createTextOutput(JSON.stringify({
    expenses: data,
    members: getMembers(ss)
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  ensureSheets(ss);

  var req = JSON.parse(e.postData.contents || '{}');

  if (req.action === 'add') {
    var sheet = ss.getSheetByName('Expenses');
    sheet.appendRow([
      req.id, req.date, req.title, Number(req.amount),
      req.paidBy, (req.involved || []).join(', '), req.createdBy || req.paidBy
    ]);
  } else if (req.action === 'delete') {
    var expSheet = ss.getSheetByName('Expenses');
    var rows = expSheet.getDataRange().getValues();
    for (var i = 1; i < rows.length; i++) {
      if (String(rows[i][0]) === String(req.id)) {
        expSheet.deleteRow(i + 1);
        break;
      }
    }
  } else if (req.action === 'replaceAllExpenses') {
    var replaceSheet = ss.getSheetByName('Expenses');
    replaceSheet.clearContents();
    replaceSheet.appendRow(['ID','Date','Title','Amount','PaidBy','Involved Members','CreatedBy']);
    (req.expenses || []).forEach(function(exp) {
      replaceSheet.appendRow([
        exp.id, exp.date, exp.title, Number(exp.amount),
        exp.paidBy, (exp.involved || []).join(', '), exp.createdBy || exp.paidBy
      ]);
    });
  } else if (req.action === 'saveMembers') {
    var memberSheet = ss.getSheetByName('Members');
    memberSheet.clearContents();
    memberSheet.appendRow(['Member Name']);
    (req.members || []).forEach(function(name) {
      if (String(name).trim()) memberSheet.appendRow([String(name).trim()]);
    });
  }

  recalculateSheetTabs(ss);
  return ContentService.createTextOutput(JSON.stringify({
    status: 'success',
    members: getMembers(ss)
  })).setMimeType(ContentService.MimeType.JSON);
}

function ensureSheets(ss) {
  var memberSheet = ss.getSheetByName('Members') || ss.insertSheet('Members');
  if (memberSheet.getLastRow() === 0) {
    memberSheet.appendRow(['Member Name']);
    ['Sai','Pavan','John','Rakesh','Praveen'].forEach(function(m) {
      memberSheet.appendRow([m]);
    });
  }

  var expSheet = ss.getSheetByName('Expenses') || ss.insertSheet('Expenses');
  if (expSheet.getLastRow() === 0) {
    expSheet.appendRow(['ID','Date','Title','Amount','PaidBy','Involved Members','CreatedBy']);
  }
}

function getMembers(ss) {
  var sheet = ss.getSheetByName('Members') || ss.insertSheet('Members');
  var rows = sheet.getDataRange().getValues();
  var members = [];
  for (var i = 1; i < rows.length; i++) {
    var name = String(rows[i][0] || '').trim();
    if (name) members.push(name);
  }
  return members;
}

function recalculateSheetTabs(ss) {
  var members = getMembers(ss);
  var expSheet = ss.getSheetByName('Expenses');
  var rows = expSheet.getDataRange().getValues();

  var totals = {};
  var shares = {};
  members.forEach(function(m) {
    totals[m] = 0;
    shares[m] = 0;
  });

  var grandTotal = 0;
  for (var i = 1; i < rows.length; i++) {
    if (!rows[i][0]) continue;

    var amt = Number(rows[i][3]) || 0;
    var payer = String(rows[i][4]).trim();
    var inv = String(rows[i][5] || '').split(',').map(function(s) {
      return s.trim();
    }).filter(Boolean);

    grandTotal += amt;
    if (totals[payer] !== undefined) totals[payer] += amt;

    if (inv.length > 0) {
      var split = amt / inv.length;
      inv.forEach(function(m) {
        if (shares[m] !== undefined) shares[m] += split;
      });
    }
  }

  var totSheet = ss.getSheetByName('Member Totals & Balances') ||
                 ss.insertSheet('Member Totals & Balances');
  totSheet.clear();
  totSheet.appendRow([
    'Member Name','Total Paid (₹)','Personal Share (₹)',
    'Net Balance (₹)','Status'
  ]);

  members.forEach(function(m) {
    var net = totals[m] - shares[m];
    var status = net > 0
      ? 'Gets Back ₹' + net.toFixed(2)
      : (net < 0 ? 'Owes ₹' + Math.abs(net).toFixed(2) : 'Settled');

    totSheet.appendRow([m, totals[m], shares[m], net, status]);
  });

  totSheet.appendRow(['GRAND TOTAL', grandTotal, grandTotal, 0, '-']);
}
